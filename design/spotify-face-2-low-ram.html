// Ultra-Low RAM Spotify Player Draw (Zero JPEG Decoder)
void renderSpotifyDisplay(const char* title, const char* artist, uint8_t eq[7], int elapsedSec, int totalSec) {
  // 1. Direct clear & top status bar (No full FB alloc)
  tft.fillRect(0, 0, 240, 20, 0x0000);
  drawVectorSpotifyLogo(8, 3, TFT_GREEN);
  tft.drawString("ESP32-DAC", 26, 6, 1);

  // 2. Reactive Equalizer Bars (7 channels via tft.fillRect)
  const int barW = 28, gap = 5;
  for (int i = 0; i < 7; i++) {
    int x = 8 + i * (barW + gap);
    drawEqualizerBarSegments(x, 26, barW, eq[i]);
  }

  // 3. Glanceable Bold Track Title (24px font vector)
  tft.setTextColor(TFT_WHITE, TFT_BLACK);
  tft.drawString(title, 8, 108, &fonts::Font4);

  // 4. Artist Subtitle in Amber/Cyan 1-bit font
  tft.setTextColor(0xFD00, TFT_BLACK); // Amber
  tft.drawString(artist, 8, 142, 2);

  // 5. 16-Segment Chunked Progress Bar & Timestamps
  renderSegmentedProgressBar(8, 194, 224, 8, elapsedSec, totalSec);
  drawTimeElapsed(elapsedSec, totalSec);
}
