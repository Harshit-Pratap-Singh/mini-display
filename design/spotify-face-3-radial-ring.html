#include <LovyanGFX.hpp>

// 3.84KB Scanline DMA Buffer (240x8 lines)
static LGFX lcd;
static LGFX_Sprite lineBuf(&lcd);

void renderSpotifyDial(float progressPct, int volume) {
  // Zero PNGs in flash: Pure trigonometric Bresenham arcs
  lcd.startWrite();
  
  // 1. Outer 360° Progress Track & Spotify Green Arc
  lcd.drawArc(120, 120, 114, 110, 0, 360, 0x2124); // Dark track
  float endAngle = 270.0f + (progressPct * 360.0f); // Starts 12 o'clock
  lcd.drawArc(120, 120, 114, 110, 270, (int)endAngle % 360, 0x1DB954);

  // 2. Progress Needle Head via Polar Coordinates
  float rad = radians(endAngle);
  int headX = 120 + 112 * cos(rad);
  int headY = 120 + 112 * sin(rad);
  lcd.fillCircle(headX, headY, 3, TFT_WHITE);

  // 3. Inner Vinyl Grooves & Volume Level Arc
  lcd.drawCircle(120, 120, 72, 0x1217);
  lcd.drawArc(120, 120, 98, 96, 120, 120 + (volume * 1.2), TFT_GREEN);

  // 4. Crisp High-Contrast Typography
  lcd.setTextDatum(MC_DATUM);
  lcd.setTextColor(TFT_WHITE, TFT_BLACK);
  lcd.drawString("GET LUCKY", 120, 115);
  lcd.setTextColor(0x1DB954, TFT_BLACK);
  lcd.drawString("DAFT PUNK", 120, 132);

  lcd.endWrite();
}
