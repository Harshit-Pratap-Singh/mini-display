// ST7789 240x240 Low-SRAM Spotify UI Engine
LGFX tft;
LGFX_Sprite lineBuf(&tft);

void renderSpotifyUI(const char* title, const char* artist, uint32_t progressMs, uint32_t totalMs) {
  // 1. Top bar: Spotify icon + Connected Device
  tft.fillRect(0, 0, 240, 20, TFT_BLACK);
  tft.setTextColor(0x1DB954, TFT_BLACK);
  tft.drawString("SPOTIFY", 16, 4, &fonts::Font0);

  // 2. Stream 80x80 RGB565 album thumbnail via DMA
  // (Decoded in 16-row chunks to save heap)
  tft.pushImageDMA(8, 26, 80, 80, albumCoverRGB565);

  // 3. Bold Track Title & Artist Strings
  tft.setTextColor(TFT_WHITE, TFT_BLACK);
  tft.setTextSize(2);
  tft.drawString(title, 96, 32);
  tft.setTextColor(0x1DB954, TFT_BLACK);
  tft.setTextSize(1);
  tft.drawString(artist, 96, 56);

  // 4. Ultra-Smooth 240px Progress Bar
  int barW = (progressMs * 224) / totalMs;
  tft.fillRect(8, 172, 224, 6, 0x222222);
  tft.fillRect(8, 172, barW, 6, 0x1DB954);

  // 5. Transport Controls: Prev, Play/Pause, Next
  tft.fillCircle(120, 210, 16, 0x1DB954);
  tft.fillRect(115, 204, 3, 12, TFT_BLACK);
  tft.fillRect(122, 204, 3, 12, TFT_BLACK);
}
